FUNCTION z_mcp_list_transports.
  " 수정 가능(Modifiable) 전송 요청 목록. 기본은 본인 요청만 조회.
  DATA lv_user TYPE e070-as4user.

  AUTHORITY-CHECK OBJECT 'S_TRANSPRT'
    ID 'TTYPE' DUMMY
    ID 'ACTVT' FIELD '03'.
  IF sy-subrc <> 0.
    RAISE not_allowed.
  ENDIF.

  lv_user = to_upper( iv_user ).
  IF lv_user IS INITIAL.
    lv_user = sy-uname.
  ENDIF.

  SELECT * FROM e070
    INTO CORRESPONDING FIELDS OF TABLE et_requests
    UP TO 100 ROWS
    WHERE trstatus = 'D'
      AND as4user  = lv_user
      AND strkorr  = space
    ORDER BY trkorr DESCENDING.
ENDFUNCTION.
