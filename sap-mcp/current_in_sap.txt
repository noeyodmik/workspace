FUNCTION z_mcp_run_payroll.
*"----------------------------------------------------------------------
*"*"로컬 인터페이스:
*"  IMPORTING
*"     VALUE(IV_PERNR) TYPE  PERNR_D OPTIONAL
*"     VALUE(IV_ABKRS) TYPE  ABKRS OPTIONAL
*"     VALUE(IV_INPER) TYPE  INPER OPTIONAL
*"     VALUE(IV_SCHEMA) TYPE  CHAR4 DEFAULT 'KR00'
*"     VALUE(IV_TEST) TYPE  XFELD DEFAULT 'X'
*"  EXPORTING
*"     VALUE(EV_OK) TYPE  XFELD
*"     VALUE(EV_MESSAGE) TYPE  STRING
*"     VALUE(EV_SEQNR) TYPE  NUMC5
*"     VALUE(EV_FPPER) TYPE  FPPER
*"     VALUE(EV_INPER) TYPE  INPER
*"     VALUE(EV_RETRO) TYPE  XFELD
*"     VALUE(EV_RETRO_COUNT) TYPE  INT4
*"     VALUE(EV_ERROR) TYPE  XFELD
*"     VALUE(EV_LOG_COUNT) TYPE  INT4
*"  TABLES
*"      ET_RT STRUCTURE  PC207 OPTIONAL
*"      ET_RETRO_PERIODS STRUCTURE  PC261 OPTIONAL
*"      ET_LOG STRUCTURE  ABAPTXT255 OPTIONAL
*"----------------------------------------------------------------------
  " 사번(PERNR) 또는 급여영역(ABKRS) 기준 한국 급여 계산(HKRCALC0 / PC00_M41_CALC) 원격 실행 및 소급/오류/RT 결과 진단
  DATA: lv_pernr        TYPE pernr_d,
        lv_inper        TYPE inper,
        lv_schema       TYPE char4,
        lv_test         TYPE xfeld,
        lv_abkrs        TYPE abkrs,
        lt_rsparams     TYPE STANDARD TABLE OF rsparams,
        lt_list         TYPE STANDARD TABLE OF abaplist,
        lt_ascii        TYPE STANDARD TABLE OF char255,
        ls_line         TYPE char255,
        lt_rgdir_before TYPE STANDARD TABLE OF pc261,
        lt_rgdir_after  TYPE STANDARD TABLE OF pc261,
        ls_rgdir        TYPE pc261,
        ls_result       TYPE paykr_result,
        lv_has_error    TYPE abap_bool VALUE abap_false,
        lv_err_msg      TYPE string.
  lv_pernr  = iv_pernr.
  lv_inper  = iv_inper.
  lv_schema = iv_schema.
  lv_test   = iv_test.
  lv_abkrs  = iv_abkrs.
  " 1. 유효성 검사: 사원번호 또는 급여영역 중 최소 하나는 지정되어야 함
  IF lv_pernr IS INITIAL AND lv_abkrs IS INITIAL.
    ev_ok = space.
    ev_message = '사원번호(IV_PERNR) 또는 급여영역(IV_ABKRS) 중 최소 하나는 입력해야 합니다.'.
    RETURN.
  ENDIF.
  DATA: lv_test_curr TYPE t500c-waers,
        lv_subrc_c   TYPE sy-subrc.
  CALL FUNCTION 'RP_GET_CURRENCY'
    EXPORTING
      molga                         = '41'
    IMPORTING
      waers                         = lv_test_curr
    EXCEPTIONS
      molga_not_in_t001p            = 1
      no_entry_found_in_table_t001  = 2
      no_entry_found_in_table_t500p = 3
      no_entry_found_in_table_t500c = 4
      OTHERS                        = 5.
  lv_subrc_c = sy-subrc.

  IF lv_schema IS INITIAL.
    lv_schema = 'KR00'.
  ENDIF.
  " 2. 권한 확인 (급여 실행 권한 또는 개발자 권한)
  AUTHORITY-CHECK OBJECT 'P_ORGIN'
    ID 'INFTY' FIELD '0008'
    ID 'SUBTY' DUMMY
    ID 'AUTHC' FIELD 'W'.
  IF sy-subrc <> 0.
    AUTHORITY-CHECK OBJECT 'S_DEVELOP'
      ID 'DEVCLASS' DUMMY
      ID 'OBJTYPE'  FIELD 'PROG'
      ID 'OBJNAME'  DUMMY
      ID 'P_GROUP'  DUMMY
      ID 'ACTVT'    FIELD '16'.
    IF sy-subrc <> 0.
      RAISE not_allowed.
    ENDIF.
  ENDIF.
  " 한국 급여 필수 인포타입 점검 및 보완 (0539 개인 데이터, 0540 세금)
  IF lv_pernr IS NOT INITIAL.
    DATA: ls_chk539 TYPE pa0539,
          ls_chk540 TYPE pa0540.
    SELECT SINGLE * FROM pa0539 INTO ls_chk539
      WHERE pernr = lv_pernr
        AND begda <= '99991231'
        AND endda >= '20200101'.
    IF sy-subrc <> 0.
      CLEAR ls_chk539.
      ls_chk539-mandt = sy-mandt.
      ls_chk539-pernr = lv_pernr.
      ls_chk539-begda = '20200101'.
      ls_chk539-endda = '99991231'.
      ls_chk539-seqnr = '000'.
      ls_chk539-aedtm = sy-datum.
      ls_chk539-uname = sy-uname.
      ls_chk539-regno = '9011111111111'.
      ls_chk539-lnmhg = '김'.
      ls_chk539-fnmhg = '검증'.
      ls_chk539-hshld = '1'.
      INSERT pa0539 FROM ls_chk539.
      COMMIT WORK.
    ENDIF.
    SELECT SINGLE * FROM pa0540 INTO ls_chk540
      WHERE pernr = lv_pernr
        AND begda <= '99991231'
        AND endda >= '20200101'.
    IF sy-subrc <> 0.
      CLEAR ls_chk540.
      ls_chk540-mandt = sy-mandt.
      ls_chk540-pernr = lv_pernr.
      ls_chk540-begda = '20200101'.
      ls_chk540-endda = '99991231'.
      ls_chk540-seqnr = '000'.
      ls_chk540-aedtm = sy-datum.
      ls_chk540-uname = sy-uname.
      ls_chk540-regno = '9011111111111'.
      ls_chk540-lnmhg = '김'.
      ls_chk540-fnmhg = '검증'.
      ls_chk540-famid = '1'.
      INSERT pa0540 FROM ls_chk540.
      COMMIT WORK.
    ENDIF.
  ENDIF.

  " 3. 단일 사원 실행 시: 사전 RGDIR 스냅샷 및 급여영역 자동 판별
  IF lv_pernr IS NOT INITIAL.
    CALL FUNCTION 'CU_READ_RGDIR'
      EXPORTING
        persnr          = lv_pernr
      TABLES
        in_rgdir        = lt_rgdir_before
      EXCEPTIONS
        no_record_found = 1
        OTHERS          = 2.
    IF lv_abkrs IS INITIAL.
      SELECT SINGLE abkrs FROM pa0001 INTO lv_abkrs
        WHERE pernr = lv_pernr
          AND begda <= sy-datum
          AND endda >= sy-datum.
      IF sy-subrc <> 0.
        SELECT abkrs FROM pa0001 INTO lv_abkrs
          UP TO 1 ROWS
          WHERE pernr = lv_pernr
          ORDER BY endda DESCENDING.
        ENDSELECT.
      ENDIF.
    ENDIF.
  ENDIF.
  " 4. HKRCALC0 SUBMIT 파라미터 구성 (LDB PNP 및 RPCFDP09 표준 파라미터 명칭)
  CLEAR lt_rsparams.

  " 사번 조건 (SELECT-OPTIONS PNPPERNR)
  IF lv_pernr IS NOT INITIAL.
    APPEND VALUE #( selname = 'PNPPERNR' kind = 'S' sign = 'I' option = 'EQ' low = lv_pernr ) TO lt_rsparams.
  ENDIF.

  " 급여 계산 영역 (LDB PNP의 최상단 필수 파라미터 PNPXABKR 및 필터 PNPABKRS)
  IF lv_abkrs IS NOT INITIAL.
    APPEND VALUE #( selname = 'PNPXABKR' kind = 'P' low = lv_abkrs ) TO lt_rsparams.
    APPEND VALUE #( selname = 'PNPABKRS' kind = 'S' sign = 'I' option = 'EQ' low = lv_abkrs ) TO lt_rsparams.
    APPEND VALUE #( selname = 'P_ABKRS'  kind = 'P' low = lv_abkrs ) TO lt_rsparams.
  ENDIF.

  " 급여 기간 조건 (지정된 경우 PNPTIMRA + 연도/기간, 미지정 시 PNPTIMR9 현재 기간)
  DATA: lv_inper_str TYPE string,
        lv_pabrj     TYPE qppnp-pabrj,
        lv_pabrp     TYPE qppnp-pabrp.
  lv_inper_str = iv_inper.
  IF lv_inper_str IS NOT INITIAL AND strlen( lv_inper_str ) >= 6.
    lv_pabrj = lv_inper_str(4).
    lv_pabrp = lv_inper_str+4(2).
    APPEND VALUE #( selname = 'PNPTIMRA' kind = 'P' low = 'X' ) TO lt_rsparams.
    APPEND VALUE #( selname = 'PNPPABRJ' kind = 'P' low = lv_pabrj ) TO lt_rsparams.
    APPEND VALUE #( selname = 'PNPPABRP' kind = 'P' low = lv_pabrp ) TO lt_rsparams.
    APPEND VALUE #( selname = 'P_INPER'  kind = 'P' low = lv_inper_str ) TO lt_rsparams.
  ELSE.
    APPEND VALUE #( selname = 'PNPTIMR9' kind = 'P' low = 'X' ) TO lt_rsparams.
  ENDIF.

  " 스키마 (PARAMETERS SCHEMA)
  APPEND VALUE #( selname = 'SCHEMA'   kind = 'P' low = lv_schema ) TO lt_rsparams.
  APPEND VALUE #( selname = 'CALCSCHM' kind = 'P' low = lv_schema ) TO lt_rsparams.

  " 테스트 실행 플래그 (PARAMETERS TST_ON / TESTRUN)
  IF lv_test = 'X'.
    APPEND VALUE #( selname = 'TST_ON'  kind = 'P' low = 'X' ) TO lt_rsparams.
    APPEND VALUE #( selname = 'TESTRUN' kind = 'P' low = 'X' ) TO lt_rsparams.
    APPEND VALUE #( selname = 'TEST'    kind = 'P' low = 'X' ) TO lt_rsparams.
  ELSE.
    APPEND VALUE #( selname = 'TST_ON'  kind = 'P' low = ' ' ) TO lt_rsparams.
    APPEND VALUE #( selname = 'TESTRUN' kind = 'P' low = ' ' ) TO lt_rsparams.
    APPEND VALUE #( selname = 'TEST'    kind = 'P' low = ' ' ) TO lt_rsparams.
  ENDIF.

  " 로그 출력 플래그 (PARAMETERS PRT_PROT / SHOWLOG)
  APPEND VALUE #( selname = 'PRT_PROT' kind = 'P' low = 'X' ) TO lt_rsparams.
  APPEND VALUE #( selname = 'SHOWLOG'  kind = 'P' low = 'X' ) TO lt_rsparams.

  " 5. HKRCALC0 호출 (출력 리스트를 메모리로 캡처)
  SUBMIT hkrcalc0
    WITH SELECTION-TABLE lt_rsparams
    EXPORTING LIST TO MEMORY
    AND RETURN.
  " 6. 메모리로부터 실행 리스트 읽기
  CALL FUNCTION 'LIST_FROM_MEMORY'
    TABLES
      listobject = lt_list
    EXCEPTIONS
      not_found  = 1
      OTHERS     = 2.
  IF sy-subrc = 0 AND lt_list IS NOT INITIAL.
    CALL FUNCTION 'LIST_TO_ASCI'
      TABLES
        listasci   = lt_ascii
        listobject = lt_list
      EXCEPTIONS
        empty_list = 1
        OTHERS     = 2.
    LOOP AT lt_ascii INTO ls_line.
      APPEND VALUE #( line = ls_line ) TO et_log.
      " 오류 / 거부 키워드 탐지
      IF ls_line CS '거부' OR ls_line CS '오류' OR ls_line CS 'Rejected' OR ls_line CS 'Error'
         OR ls_line CS 'Fehler' OR ls_line CS 'abgelehnt'.
        IF NOT ls_line CS '0 오류' AND NOT ls_line CS '0 거부'.
          lv_has_error = abap_true.
          IF lv_err_msg IS INITIAL.
            lv_err_msg = ls_line.
          ENDIF.
        ENDIF.
      ENDIF.
    ENDLOOP.
  ENDIF.
  ev_log_count = lines( et_log ).
  " 7. 결과 분석
  IF lv_has_error = abap_true.
    ev_error = 'X'.
    ev_ok    = space.
    ev_message = |[CURR_CHK subrc={ lv_subrc_c }, curr={ lv_test_curr }] 급여 계산 중 오류가 발생했습니다: { lv_err_msg }|.
    RETURN.
  ENDIF.
  IF lv_test = 'X'.
    " 시뮬레이션(테스트 실행) 완료
    ev_ok = 'X'.
    IF lv_pernr IS NOT INITIAL.
      ev_message = |시뮬레이션(테스트 실행) 완료 (사번: { lv_pernr }, 스키마: { lv_schema })|.
    ELSE.
      ev_message = |시뮬레이션(테스트 실행) 완료 (급여영역: { lv_abkrs }, 스키마: { lv_schema })|.
    ENDIF.
    RETURN.
  ENDIF.
  " 단일 사원이 아닌 급여영역 전체 일괄 실행인 경우 로그 요약 반환
  IF lv_pernr IS INITIAL.
    ev_ok = 'X'.
    ev_message = |급여영역 { lv_abkrs } 실 급여 계산 완료 (처리 로그 { ev_log_count }건 생성)|.
    RETURN.
  ENDIF.
  " 단일 사원 실제 실행인 경우 DB 클러스터에서 결과 및 소급 분석
  CALL FUNCTION 'CU_READ_RGDIR'
    EXPORTING
      persnr          = lv_pernr
    TABLES
      in_rgdir        = lt_rgdir_after
    EXCEPTIONS
      no_record_found = 1
      OTHERS          = 2.
  IF sy-subrc <> 0 OR lt_rgdir_after IS INITIAL.
    ev_ok = 'X'.
    ev_message = |급여 실행이 완료되었으나 유효한 결과 레코드가 생성되지 않았습니다.|.
    RETURN.
  ENDIF.
  DELETE lt_rgdir_after WHERE void IS NOT INITIAL.
  SORT lt_rgdir_after BY seqnr DESCENDING.
  " 최신 결과 레코드
  READ TABLE lt_rgdir_after INTO ls_rgdir WITH KEY srtza = 'A'.
  IF sy-subrc <> 0.
    READ TABLE lt_rgdir_after INTO ls_rgdir INDEX 1.
  ENDIF.
  ev_seqnr = ls_rgdir-seqnr.
  ev_fpper = ls_rgdir-fpper.
  ev_inper = ls_rgdir-inper.
  " 소급(Retrocalculation) 탐지: 이번 실행(동일 inper) 중 fpper < inper 인 레코드 또는 이전 결과 대체 건 탐지
  LOOP AT lt_rgdir_after INTO DATA(ls_check) WHERE inper = ls_rgdir-inper.
    IF ls_check-fpper < ls_check-inper OR ls_check-srtza = 'P'.
      ev_retro = 'X'.
      APPEND ls_check TO et_retro_periods.
    ENDIF.
  ENDLOOP.
  ev_retro_count = lines( et_retro_periods ).
  " 8. 실제 계산된 RT(Results Table) 읽기 및 전달
  CALL FUNCTION 'PYXX_READ_PAYROLL_RESULT'
    EXPORTING
      clusterid                   = 'KR'
      employeenumber              = lv_pernr
      sequencenumber              = ls_rgdir-seqnr
    CHANGING
      payroll_result              = ls_result
    EXCEPTIONS
      illegal_is_seqnr            = 1
      not_found                   = 2
      country_version_not_matched = 3
      OTHERS                      = 4.
  IF sy-subrc = 0.
    et_rt[] = ls_result-inter-rt.
  ENDIF.
  ev_ok = 'X'.
  IF ev_retro = 'X'.
    ev_message = |실 급여 계산 완료 (사번: { lv_pernr }, 순번: { ls_rgdir-seqnr }, 귀속: { ls_rgdir-fpper }, 소급 { ev_retro_count }개 기간 발생, { lines( et_rt ) }개 임금유형 산출)|.
  ELSE.
    ev_message = |실 급여 계산 완료 (사번: { lv_pernr }, 순번: { ls_rgdir-seqnr }, 귀속: { ls_rgdir-fpper }, { lines( et_rt ) }개 임금유형 산출)|.
  ENDIF.
ENDFUNCTION.
