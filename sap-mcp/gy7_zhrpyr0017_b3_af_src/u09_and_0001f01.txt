FUNCTION ZHR0004_GET_PAYRESULT.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(PERNR) TYPE  PERSNO
*"     VALUE(ABKRS) TYPE  ABKRS
*"     VALUE(YYYYMM) TYPE  FAPER
*"     VALUE(OCRSN) TYPE  PAY_OCRSN OPTIONAL
*"     VALUE(PAYID) TYPE  PAYID OPTIONAL
*"     VALUE(PAYDT) TYPE  PAY_DATE
*"     VALUE(RETRO) TYPE  ZRETRO DEFAULT 'SUM'
*"     VALUE(FLAG) TYPE  CHAR01 DEFAULT 'X'
*"  EXPORTING
*"     VALUE(RESULT) LIKE  ZHRS0012 STRUCTURE  ZHRS0012
*"     VALUE(EX_PAYDT) TYPE  PAYDT
*"  EXCEPTIONS
*"      SYSTEM_FAILURE
*"      COMMUNICATION_FAILURE
*"      RESOURCE_FAILURE
*"----------------------------------------------------------------------
*                           Change History
*---------- ------------ ----------- -----------------------------------
* ChangeNo.  Changed on   Changed By  Description
*---------- ------------ ----------- -----------------------------------
* 1426        2023.08.26  황용하      코비전 I/F 로그 저장 기능 추가
* 5369        2024.04.30  황용하      급여지급대장 수정
* 5390        2024.05.30  황용하      특별성과급(비과세)를 비과세소득에 합산
* 5951        2024.06.19  황용하      임금유형:휴가비 추가에 따른 수정
************************************************************************
  DATA: PAY_DATE  TYPE PAY_DATE,
        RETROYM   TYPE FAPER,
        PAY_BEGDA TYPE BEGDA,
        PAY_ENDDA TYPE ENDDA.
  DATA: LV_DATE     TYPE SY-DATUM.
  DATA: LS_STR      TYPE ZHRS0004.

  DATA: LT_P2003    TYPE TABLE OF P2003 WITH HEADER LINE.
*----------------------------------------------------------------------*
* 급여결과 가져오기
*----------------------------------------------------------------------*
  CLEAR: RT[], CRT[], WPBP[].
  CALL FUNCTION 'ZHR0004_PYKRF001'
    EXPORTING
      PERNR     = PERNR
      ABKRS     = ABKRS
      YYYYMM    = YYYYMM
      OCRSN     = OCRSN
      PAYID     = PAYID
      PAYDT     = PAYDT
      RETRO     = RETRO
      FLAG      = FLAG
    IMPORTING
      PAY_DATE  = PAY_DATE
      RETROYM   = RETROYM
      PAY_BEGDA = PAY_BEGDA
      PAY_ENDDA = PAY_ENDDA
    TABLES
      RT        = RT
      CRT       = CRT
      WPBP      = WPBP
      AB        = AB.

  EX_PAYDT = PAY_DATE.

*"BETRG(금액), BETPE(단가) 필드 모두 값이 없는 레코드 삭제
  DELETE RT WHERE BETRG IS INITIAL
              AND BETPE IS INITIAL
              AND ANZHL IS INITIAL.

  CHECK RT[] IS NOT INITIAL.

  IF  RT[]   IS NOT INITIAL.
    LOOP   AT RT.
      IF     OCRSN  =  ' '
        OR OCRSN  =  '0007'
        OR OCRSN  =  '0008'
        OR OCRSN  =  '0005'
        OR OCRSN  =  '9001'.
        PERFORM GET_REGULAR USING RT.
      ELSE.
        PERFORM GET_BONUS   USING RT.
      ENDIF.
    ENDLOOP.
  ENDIF.

* 지급금액 합산에 기지급금 제외 (P26 ~ P30)
  WA-P_TOTAL = WA-P01  + WA-P02  + WA-P03  + WA-P04  + WA-P05
             + WA-P06  + WA-P07  + WA-P08  + WA-P09  + WA-P10
             + WA-P11  + WA-P12  + WA-P13  + WA-P14  + WA-P15
             + WA-P16  + WA-P17  + WA-P18  + WA-P19  + WA-P20
             + WA-P21  + WA-P22  + WA-P23  + WA-P25  + WA-P27
             + WA-P28   "20250515 추가 (기타수당)
             + WA-P29   "20260901 추가 (포상금) SEPPIN KDY
             + WA-P32   "20260325 추가 (AI수당)
             + WA-P26  + WA-R_TOTAL.                           " #5951 추가

  "공제총액
  WA-D_TOTAL = WA-D01_2 + WA-D02_2 + WA-D03 + WA-D04_2 + WA-D05_2
             + WA-D06_2
             + WA-D07   + WA-D08   + WA-D09 + WA-D10
             + WA-D11   + WA-D12   + WA-D13 + WA-D14   + WA-D15
             + WA-D16   + WA-D17.

  "실지급액
  WA-TOTAL = WA-P_TOTAL - WA-D_TOTAL. "실지급액

  "회사지급합계
  WA-C_TOTAL = WA-C01   + WA-C02   + WA-C03  + WA-C04  + WA-C05.

  "근무일수가 없으면 총근로시간수도 조회하지 않는다.
  CLEAR WA-H00.
  LOOP AT WPBP.
    WA-H00 = WA-H00 + WPBP-ASOLL.
  ENDLOOP.

  IF   WA-H00 = 0.
    CLEAR WA-H99.
  ELSE.
    WA-H99 = 209 + WA-T02 + WA-T03 + WA-T04 + WA-T05 + WA-T06.

  ENDIF.


  MOVE-CORRESPONDING WA TO RESULT.

  RESULT-FPPER = YYYYMM.
  RESULT-WAERS = 'KRW'.
  RESULT-PAYDT = PAY_DATE.
  RESULT-OCRSN = OCRSN.
  RESULT-PAYID = PAYID.


**----------------------------------------------------------------------*
** 휴무 정보 가져오기
**----------------------------------------------------------------------*
  HD-H00 = WA-H00.
  HD-H99 = WA-H99.
  CLEAR WA.
  IF  AB[]  IS  NOT  INITIAL.
    LOOP  AT  AB.
      IF  OCRSN  = ' '
      OR  OCRSN  = '0007'.
        PERFORM   GET_HOLIDAYS USING AB.
      ENDIF.
    ENDLOOP.
  ENDIF.

**----------------------------------------------------------------------*
** 근무 정보 가져오기(IT2003)
**----------------------------------------------------------------------*
*  CLEAR: LT_P2003, LT_P2003[].
*
*  SELECT * INTO CORRESPONDING FIELDS OF TABLE LT_P2003
*    FROM PA2003
*   WHERE PERNR = PERNR
*     AND BEGDA  >= PAY_BEGDA
*     AND ENDDA  <= PAY_ENDDA.
*
*  IF LT_P2003[] IS NOT INITIAL.
*    LOOP AT LT_P2003.
*      HD-H00  =   HD-H00  +  1 .  "근무일수
*      HD-H99  =   HD-H99  +  LT_P2003-STDAZ .  "근무시간
*    ENDLOOP.
*  ENDIF.
*

  MOVE-CORRESPONDING HD TO RESULT .

  CLEAR HD.
*----------------------------------------------------------------------*
* 인사정보 가져오기
*----------------------------------------------------------------------*

*  IF   PAY_DATE IS NOT INITIAL.
*    LV_DATE = PAY_DATE.
*  ELSE.
*    CONCATENATE YYYYMM '01' INTO LV_DATE.
*  ENDIF.

  CLEAR LV_DATE.
  CONCATENATE YYYYMM '01' INTO LV_DATE.

  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      DAY_IN                  = LV_DATE
   IMPORTING
     LAST_DAY_OF_MONTH       = LV_DATE.

  CALL FUNCTION 'ZHR0002_GET_BASIC_DATA'
    EXPORTING
      IM_DATE  = LV_DATE
      IM_PERNR = PERNR
    IMPORTING
      EX_STRU  = LS_STR.

  MOVE-CORRESPONDING LS_STR TO RESULT.
  MOVE PAY_DATE TO RESULT-PAYDT. "지급일

ENDFUNCTION.

    EXPORTING
      IT_FIELDCAT_ALV = GT_FIELDCAT_ALV[]
    IMPORTING
      ET_FIELDCAT_LVC = <FIELDCAT>
    TABLES
      IT_DATA         = <FCAT>     "gt_dis
    EXCEPTIONS
      IT_DATA_MISSING = 1
      OTHERS          = 2.

    CHANGING

FORM SAVE_DATA .

  DATA: IT_HRPY_RGDIR TYPE HRPY_RGDIR OCCURS 0 WITH HEADER LINE,
        LV_BEGDA      TYPE BEGDA,
        LV_ENDDA      TYPE ENDDA.

  CALL FUNCTION 'RP_CALC_DATE_IN_INTERVAL'
    EXPORTING
      DATE      = SY-DATUM
      DAYS      = '00'
      MONTHS    = '01'
      SIGNUM    = '-'
      YEARS     = '00'
    IMPORTING
      CALC_DATE = LV_BEGDA. "전달의 마지막일

  LV_BEGDA+6(2) = '01'.

  CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
    EXPORTING
      DAY_IN            = LV_BEGDA
    IMPORTING
      LAST_DAY_OF_MONTH = LV_ENDDA.



  CLEAR : TAB_LINE.
  DESCRIBE TABLE GT_PERNR LINES TAB_LINE.

*----------------------------------------------------------------------*
* 급여결과 가져오기 - 병렬처리
*----------------------------------------------------------------------*
  LOOP AT GT_PERNR.

    "현재날자기준으로 데이터생성여부판단
    CLEAR : IT_HRPY_RGDIR, IT_HRPY_RGDIR[].
    SELECT * INTO CORRESPONDING FIELDS OF TABLE IT_HRPY_RGDIR
      FROM HRPY_RGDIR
     WHERE PERNR = GT_PERNR-PERNR
       AND ( ( FPPER = LV_BEGDA+0(6) AND OCRSN IN ('','0007') )
        OR PAYDT BETWEEN LV_BEGDA AND LV_ENDDA
       AND OCRSN IN ('0002','0003','0004') ).

    SORT : IT_HRPY_RGDIR  BY  PAYDT OCRSN PAYID.
    DELETE ADJACENT DUPLICATES FROM IT_HRPY_RGDIR
     COMPARING PAYDT OCRSN PAYID.
    DELETE IT_HRPY_RGDIR WHERE SRTZA <> 'A'.

    LOOP AT IT_HRPY_RGDIR.
      CLEAR: PNPPABRJ, PNPPABRP, V_YYYYMM, P_OCRSN, P_PAYID, P_PAYDT.
      IF IT_HRPY_RGDIR-OCRSN IS INITIAL OR IT_HRPY_RGDIR-OCRSN = '0007'.
        PNPPABRJ = IT_HRPY_RGDIR-FPPER+0(4).
        PNPPABRP = IT_HRPY_RGDIR-FPPER+4(2).
        CONCATENATE PNPPABRJ PNPPABRP INTO V_YYYYMM.
        ITAB-ZSTYPE = 100.
      ELSE.
        V_YYYYMM    = IT_HRPY_RGDIR-PAYDT.
        ITAB-ZSTYPE = 200.
      ENDIF.

      P_OCRSN       = IT_HRPY_RGDIR-OCRSN.
      P_PAYID       = IT_HRPY_RGDIR-PAYID.
      P_PAYDT       = IT_HRPY_RGDIR-PAYDT.

      IF       S1   = 'X'.            "소급분별도
        LV_RETRO  = 'NOW'.
      ELSEIF   S2   = 'X'.            "소급분포함
        LV_RETRO  = 'SUM'.
      ELSEIF   S3   = 'X'.            "소급분만
        LV_RETRO  = 'RET'.
      ENDIF.

      WAIT UNTIL G_PROCS <= P_PROCS.

      TEMP_TEXT = '진행중'.
      REC_COUNT = REC_COUNT + 1.
      PCT = REC_COUNT * 100 / TAB_LINE.

      PERFORM PROGRESS_INDICATOR  USING REC_COUNT
                                        TAB_LINE
                                        TEMP_TEXT.

      CLEAR GS_RESULT.
      CALL FUNCTION 'ZHR0004_GET_PAYRESULT'
        STARTING NEW TASK TASKNAME
        DESTINATION IN GROUP DEFAULT
        PERFORMING RECEIVE ON END OF TASK
        EXPORTING
          PERNR                 = GT_PERNR-PERNR
          ABKRS                 = GT_PERNR-ABKRS
          YYYYMM                = V_YYYYMM
          OCRSN                 = P_OCRSN
          PAYID                 = P_PAYID
          PAYDT                 = P_PAYDT
          RETRO                 = LV_RETRO
        EXCEPTIONS
          SYSTEM_FAILURE        = 1
          COMMUNICATION_FAILURE = 2
          RESOURCE_FAILURE      = 3.


      CASE SY-SUBRC.
        WHEN 0.
          TASKNAME = TASKNAME + 1.
          SND_JOBS = SND_JOBS + 1.
          G_PROCS  = G_PROCS + 1.
        WHEN 1 OR 2.
        WHEN 3.
          IF EXCP_FLAG = SPACE.
            EXCP_FLAG = 'X'.
            WAIT UNTIL RCV_JOBS >= SND_JOBS UP TO '1' SECONDS.
          ELSE.
            WAIT UNTIL RCV_JOBS >= SND_JOBS UP TO '5' SECONDS.
            IF SY-SUBRC = 0.
              CLEAR EXCP_FLAG.
            ELSE.  "No replies
            ENDIF.
          ENDIF.
      ENDCASE.

    ENDLOOP.

    WAIT UNTIL RCV_JOBS >= SND_JOBS.


    SORT ITAB ASCENDING BY ORGEH PERNR.

  ENDLOOP.


ENDFORM.                    " SAVE_DATA

        LT_RETURN  LIKE TABLE OF BAPIRET2.

  REFRESH: LT_PROFILE[], LT_RETURN[].

  "실행 대상자 권한 목록 확인
  CALL FUNCTION 'BAPI_USER_GET_DETAIL'
    EXPORTING
      USERNAME      = SY-UNAME
      CACHE_RESULTS = 'X'
    TABLES
      PROFILES      = LT_PROFILE[]
      RETURN        = LT_RETURN[].

  CHECK LT_PROFILE[] IS NOT INITIAL.

  "권한 목록 중 'ECP 담당자 권한'이 없을 경우 급여 영역 편집 불가
  READ TABLE LT_PROFILE ASSIGNING FIELD-SYMBOL(<FS_PROFILE>) WITH KEY BAPIPROF = 'T-G7520019'.
  IF SY-SUBRC NE 0.
    LOOP AT SCREEN.
      CASE SCREEN-NAME.
        WHEN 'PNPXABKR' OR 'PNPABKRS-LOW'.
          SCREEN-INPUT = 0.
      ENDCASE.

      MODIFY SCREEN.
    ENDLOOP.
  ENDIF.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  MODIFY_SELECTION_SCREEN
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
FORM MODIFY_SELECTION_SCREEN .

  IF P_HIS = 'X'.
    GV_B9TXT = '저장결과 조회조건(필수)'.
    R1 = 'X'.
    CLEAR: R2,
           P9,
           P10.
  ELSE.
    GV_B9TXT = '급여결과 저장(급여명세서)'.
  ENDIF.

  LOOP AT SCREEN.
    IF P_HIS = 'X'.
      CASE SCREEN-GROUP1.
        WHEN 'OUT'.
          SCREEN-ACTIVE = 0.
        WHEN 'RET'.
          SCREEN-ACTIVE = 0.
        WHEN 'SAV'.
          SCREEN-ACTIVE = 0.
      ENDCASE.

    ELSE.
      CASE SCREEN-GROUP1.
        WHEN 'OUT' OR 'RET' OR 'SAV'.
          SCREEN-ACTIVE = 1.
      ENDCASE.
    ENDIF.
    MODIFY SCREEN.
  ENDLOOP.

ENDFORM.
*&---------------------------------------------------------------------*
*&      Form  GET_SAVED_DATA
*&---------------------------------------------------------------------*
*       text
*----------------------------------------------------------------------*
*  -->  p1        text
*  <--  p2        text
*----------------------------------------------------------------------*
* ZHRT0100에 저장된 급여결과 조회   2026.09.02 BY SEPPIN LJS
* 급여기간, 급여유형, 실지급일 기준으로 저장결과를 조회하고
* 실지급일 기준 인사정보를 조회하여 ALV 출력 데이터를 구성
*----------------------------------------------------------------------*
FORM GET_SAVED_DATA .

  DATA: LT_ZHRT0100 TYPE TABLE OF ZHRT0100,
        LS_ZHRT0100 TYPE ZHRT0100,
        LS_BASIC    TYPE ZHRS0004,
        LV_PERNR    TYPE PERNR_D.

  CLEAR: ITAB, ITAB[].
  CLEAR: LT_ZHRT0100, LT_ZHRT0100[].

* 대상 사원 확인
  CHECK GT_PERNR[] IS NOT INITIAL.

  SELECT *
    INTO TABLE LT_ZHRT0100
    FROM ZHRT0100
    FOR ALL ENTRIES IN GT_PERNR
   WHERE PERNR  = GT_PERNR-PERNR
     AND PABRJ  = PNPPABRJ
     AND PABRP  = PNPPABRP
     AND ZSTYPE = P_ZSTYPE
     AND ZPAYDT = P_ZPAYDT.

* 비정기상여 조회조건
  "급여사유
  IF P_OCRSN IS NOT INITIAL.
    DELETE LT_ZHRT0100
     WHERE OCRSN <> P_OCRSN.
  ENDIF.

  "지급일자
  IF P_PAYDT IS NOT INITIAL.
    DELETE LT_ZHRT0100
     WHERE PAYDT <> P_PAYDT.
  ENDIF.

  "급여계산 식별자
  IF P_PAYID IS NOT INITIAL.
    DELETE LT_ZHRT0100
     WHERE PAYID <> P_PAYID.
  ENDIF.

  LOOP AT LT_ZHRT0100 INTO LS_ZHRT0100.

    CLEAR: ITAB, LS_BASIC, LV_PERNR.

    MOVE-CORRESPONDING LS_ZHRT0100 TO ITAB.

* 인사정보 가져오기
    LV_PERNR = LS_ZHRT0100-PERNR.

    CALL FUNCTION 'ZHR0002_GET_BASIC_DATA'
      EXPORTING
        IM_DATE  = P_ZPAYDT    "실지급일
        IM_PERNR = LV_PERNR
      IMPORTING
        EX_STRU  = LS_BASIC.

    MOVE-CORRESPONDING LS_BASIC TO ITAB.

* 실지급일 기준 직책명 조회
    SELECT SINGLE ZZPSGRPTX
      INTO ITAB-POSNAME
      FROM ZHRT0003
     WHERE SPRAS = '3'
       AND ZZPSGRP IN ( SELECT ZZPSGRP
                          FROM PA0001
                         WHERE PERNR = ITAB-PERNR
                           AND BEGDA <= P_ZPAYDT
                           AND ENDDA >= P_ZPAYDT ).

    APPEND ITAB.

  ENDLOOP.

  SORT ITAB ASCENDING BY ORGEH PERNR.

ENDFORM.
