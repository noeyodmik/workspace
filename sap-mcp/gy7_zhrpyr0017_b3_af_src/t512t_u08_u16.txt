FUNCTION ZHR0004_GET_PAYRESULT_V2.
*"--------------------------------------------------------------------
*"*"로컬인터페이스:
*"  IMPORTING
*"     VALUE(PERNR) TYPE  PERSNO
*"     VALUE(ABKRS) TYPE  ABKRS
*"     VALUE(YYYYMM) TYPE  FAPER OPTIONAL
*"     VALUE(OCRSN) TYPE  PAY_OCRSN OPTIONAL
*"     VALUE(PAYID) TYPE  PAYID OPTIONAL
*"     VALUE(PAYDT) TYPE  PAY_DATE OPTIONAL
*"     VALUE(MODE) TYPE  CHAR01
*"     VALUE(FLAG) TYPE  CHAR01 DEFAULT 'X'
*"  EXPORTING
*"     REFERENCE(PAY_DATE) TYPE  PAY_DATE
*"     REFERENCE(RETROYM) TYPE  FAPER
*"  TABLES
*"      RT STRUCTURE  ZHRS0013 OPTIONAL
*"      CRT STRUCTURE  PC208 OPTIONAL
*"      BT STRUCTURE  PC209 OPTIONAL
*"      WPBP STRUCTURE  PC205 OPTIONAL
*"      KRSI STRUCTURE  PC2H3 OPTIONAL
*"--------------------------------------------------------------------
** 정규급여시 급여일자 셋팅
  DATA: LV_BEGDA      TYPE BEGDA,
        LV_ENDDA      TYPE ENDDA.
  DATA: LS_T549A      TYPE T549A.
  DATA: PABRJ         TYPE PABRJ,
        PABRP         TYPE PABRP.

  DATA: PRINT_START(6),
        PRINT_END(6).
  DATA: NUM           TYPE I.
  DATA: LT_RT         TYPE TABLE OF ZHRS0013  WITH HEADER LINE.

  CLEAR: RT, CRT, WPBP.
  REFRESH: RT, CRT, WPBP.


  IF OCRSN  =   ''.
    MOVE YYYYMM+0(4)  TO PABRJ.
    MOVE YYYYMM+4(2)  TO PABRP.
*    CALL FUNCTION 'HR_GB_PAY_DATE'
*      EXPORTING
*        MOLGA              = '41'
*        ABKRS              = ABKRS
*        PABRJ              = PABRJ
*        PABRP              = PABRP
*      IMPORTING
*        PDATE              = PAY_DATE
*      EXCEPTIONS
*        PAY_DATE_NOT_FOUND = 1
*        OTHERS             = 2.

    SELECT SINGLE * INTO CORRESPONDING FIELDS OF LS_T549A
      FROM T549A
     WHERE ABKRS = ABKRS.

    "급여기간 가져오기
    CALL FUNCTION 'HR_GB_PERIOD_DATES'
      EXPORTING
        ABKRS            = ABKRS
        PERMO            = LS_T549A-PERMO
        PABRJ            = PABRJ
        PABRP            = PABRP
      IMPORTING
        BEGDA            = LV_BEGDA
        ENDDA            = LV_ENDDA
      EXCEPTIONS
        PERIOD_NOT_FOUND = 1
        OTHERS           = 2.

*직전 급여기간 마지막 일자 가져오기 (전월 0008 퇴직정산)
    CALL FUNCTION 'MONTH_PLUS_DETERMINE'
      EXPORTING
        MONTHS  = '-1'
        OLDDATE = LV_ENDDA
      IMPORTING
        NEWDATE = TMP_DATE1
      EXCEPTIONS
        OTHERS  = 1.

    CALL FUNCTION 'RP_LAST_DAY_OF_MONTHS'
      EXPORTING
        DAY_IN            = TMP_DATE1
      IMPORTING
        LAST_DAY_OF_MONTH = TMP_DATE1.

  ENDIF.


* 모든급여 data을 읽어온다.
  CLEAR: RGDIR, RGDIR[].
  CALL FUNCTION 'CU_READ_RGDIR'
    EXPORTING
      PERSNR          = PERNR
    TABLES
      IN_RGDIR        = RGDIR        "structure pc261.
    EXCEPTIONS
      NO_RECORD_FOUND = 1
      OTHERS          = 2.

  IF SY-SUBRC = 1.
  ENDIF.

  W_PERNR = PERNR.


  CLEAR: LT_RT[], CRT[], BT[], WPBP[], KRSI[].

* 급여사유별 급여결과 취득
  CASE OCRSN.
    WHEN '0007' OR ''.             "정기급여
      PERNR_ABKRS = ABKRS.
*      W_RETRO     = RETRO.
      GV_MODE     = MODE.
      PN_PER      = YYYYMM.
      PRINT_START = PN_PER.
      PRINT_END   = PN_PER.
      CHECK RGDIR-OCRSN EQ ''.

      IF RGDIR-INPER EQ RGDIR-FPPER AND
         RGDIR-OCRSN EQ RGDIR-INOCR.
        PERFORM GET_RESULT_FROM_RETO_2 TABLES RGDIR
                                              LT_RT
                                              CRT
                                              BT
                                              WPBP
                                              KRSI
                                        USING PRINT_START
                                              PRINT_END
                                              FLAG
                                     CHANGING RETROYM.
        PAY_DATE = RGDIR-PAYDT.
      ENDIF.


    WHEN '0005'.                 "연말정산.
      LOOP AT RGDIR.
        CHECK RGDIR-PAYDT EQ PAYDT.
        CHECK RGDIR-OCRSN EQ '0005'.
        IF RGDIR-SRTZA EQ 'A'.
          PERFORM GET_RESULTS_2 TABLES LT_RT
                                       CRT
                                       BT
                                       WPBP
                                 USING RGDIR.
          PAY_DATE = RGDIR-PAYDT.
        ENDIF.
      ENDLOOP.


    WHEN OTHERS.                 "그 외 비정기
      LOOP AT RGDIR.
        CHECK RGDIR-PAYDT EQ PAYDT.
        CHECK RGDIR-OCRSN EQ OCRSN.
        CHECK RGDIR-PAYID EQ PAYID.

        IF RGDIR-INPER EQ RGDIR-FPPER AND
           RGDIR-OCRSN EQ RGDIR-INOCR.
          PERFORM GET_RESULTS_2 TABLES LT_RT
                                       CRT
                                       BT
                                       WPBP
                                 USING RGDIR.
          PAY_DATE = RGDIR-PAYDT.
        ENDIF.
      ENDLOOP.

  ENDCASE.

* 각 조회옵션별 임금유형 collect를 위해 필드값 클리어
  LOOP AT LT_RT.
    CASE MODE.
      WHEN 'A'.  "당월+소급분합산
        CLEAR: LT_RT-FPPER,     "대상기간
               LT_RT-INPER,     "조회기간
               LT_RT-PAYID,     "식별자
               LT_RT-PAYDT,     "급여 지급일
               LT_RT-OCRSN.     "급여사유

      WHEN 'B'.  "당월+소급분 월별
        CLEAR: LT_RT-PAYID,     "식별자
               LT_RT-PAYDT,     "급여 지급일
               LT_RT-OCRSN.     "급여사유

      WHEN 'C'.  "당월+소급분지급분별

      WHEN 'D'.  "당월분만
        IF LT_RT-FPPER <> LT_RT-INPER.
          CLEAR: LT_RT-FPPER,     "대상기간
                 LT_RT-INPER,     "조회기간
                 LT_RT-PAYID,     "식별자
                 LT_RT-PAYDT,     "급여 지급일
                 LT_RT-OCRSN.     "급여사유

          DELETE TABLE LT_RT FROM LT_RT.
          CONTINUE.
        ENDIF.

      WHEN 'E'.  "소급분만 합산
        IF LT_RT-FPPER = LT_RT-INPER.
          CLEAR: LT_RT-FPPER,     "대상기간
                 LT_RT-PAYID,     "식별자
                 LT_RT-PAYDT,     "급여 지급일
                 LT_RT-OCRSN.     "급여사유

          DELETE TABLE LT_RT FROM LT_RT.
          CONTINUE.
        ENDIF.

      WHEN 'F'. "소급분지급분별
        IF LT_RT-FPPER = LT_RT-INPER.
          DELETE TABLE LT_RT FROM LT_RT.
          CONTINUE.
        ENDIF.

      WHEN 'G'. "당월분 + 소급분별도

    ENDCASE.

    MOVE-CORRESPONDING LT_RT TO RT.
    COLLECT RT.

  ENDLOOP.





ENDFUNCTION.
=== LZHR0004U16 (ZHR0004_GET_ANNUAL_REWARD) ===
FUNCTION ZHR0004_GET_ANNUAL_REWARD.
*"----------------------------------------------------------------------
*"*"Local interface:
*"  IMPORTING
*"     VALUE(IM_PERNR) TYPE  PERSNO OPTIONAL
*"     VALUE(IM_DATE) TYPE  DATS
*"  EXPORTING
*"     VALUE(E_CODE) TYPE  CHAR01
*"     VALUE(E_MSG) TYPE  BAPI_MSG
*"  TABLES
*"      T_TABLE STRUCTURE  ZHRS0039
*"----------------------------------------------------------------------

  DATA: LT_ZHRPT0002        TYPE TABLE OF ZHRPT0002 WITH HEADER LINE.
  DATA: LT_ZHRPT0002_O    TYPE TABLE OF ZHRPT0002 WITH HEADER LINE.
  DATA: LT_KEY                  LIKE TABLE OF LT_ZHRPT0002 WITH HEADER LINE.
  DATA: LV_CHK TYPE CHAR01.

  DATA: LV_DATE TYPE SY-DATUM,
            LV_ZEIT TYPE SY-UZEIT.

  CLEAR:  LT_ZHRPT0002, LT_ZHRPT0002_O.

* log 테이블에 저장   =================================
  PERFORM make_logkey.

  zhriflog-itype = 'AR'.
  zhriflog-param = im_pernr && '/' && im_DATE.

  LOOP AT t_table.
    ADD 1 TO zhriflog-seqno.
    zhriflog-data =  t_table-pernr && '/' && t_table-ANZHL && '/' && t_table-LEGZHL.
    INSERT zhriflog.
  ENDLOOP.
  zhriflog-seqno = 0.
  zhriflog-data  = ''.
*======================================================

  SELECT *
    INTO CORRESPONDING FIELDS OF TABLE LT_ZHRPT0002_O
    FROM ZHRPT0002.

  SORT LT_ZHRPT0002_O BY PERNR BEGDA.

  IF im_pernr IS NOT INITIAL. " 사원번호가 import 될 경우 해당 사원 외는 삭제
    DELETE t_table WHERE pernr <> im_pernr.
  ENDIF.

  " 보상일 체크

  IF IM_DATE IS INITIAL.
    E_CODE = 'E'.   " 에러.
    E_MSG = '보상일을 입력해 주십시오'.

    zhriflog-param = 'E1_' && E_MSG.
    insert zhriflog.
    RETURN.
  ENDIF.

  LOOP AT T_TABLE.

    CLEAR: LT_ZHRPT0002, LV_DATE, LV_ZEIT.

    AT NEW PERNR.
      CLEAR: LV_CHK.
    ENDAT.

    LT_ZHRPT0002-PERNR = T_TABLE-PERNR.
    LT_ZHRPT0002-BEGDA = IM_DATE.
    LT_ZHRPT0002-ANZHL = T_TABLE-ANZHL.
    LT_ZHRPT0002-LEGZHL = T_TABLE-LEGZHL.
    LT_ZHRPT0002-LGART = '3700'.


    LT_ZHRPT0002-ERNAM = SY-UNAME.

    CALL FUNCTION 'ISU_DATE_TIME_CONVERT_TIMEZONE'
      EXPORTING
        X_DATE_UTC = SY-DATUM
        X_TIME_UTC = SY-UZEIT
        X_TIMEZONE = SY-ZONLO
      IMPORTING
        Y_DATE_LCL = LV_DATE
        Y_TIME_LCL = LV_ZEIT.

    LT_ZHRPT0002-ERDAT = LV_DATE.
    LT_ZHRPT0002-ERTIM = LV_ZEIT.

    APPEND LT_ZHRPT0002.

    IF LV_CHK = 'X'.
      CONTINUE.
    ENDIF.

    READ TABLE LT_ZHRPT0002_O
          TRANSPORTING NO FIELDS
            WITH KEY BEGDA+0(4) = IM_DATE+0(4).

    IF SY-SUBRC = 0.

      LOOP AT LT_ZHRPT0002_O FROM SY-TABIX WHERE BEGDA+0(4) = IM_DATE+0(4).

        LV_CHK = 'X'.

        MOVE-CORRESPONDING LT_ZHRPT0002_O TO LT_KEY.

        COLLECT LT_KEY. CLEAR LT_KEY.

      ENDLOOP.

    ENDIF.

  ENDLOOP.

  LOOP AT LT_KEY.

    DELETE LT_ZHRPT0002_O WHERE PERNR = LT_KEY-PERNR
                                               AND BEGDA = LT_KEY-BEGDA.

  ENDLOOP.

  IF LT_ZHRPT0002[] IS INITIAL.
    E_CODE = 'E'.   " 오류.
    E_MSG = '입력값을 확인해 주십시오'.

    zhriflog-param = 'E1_' && E_MSG.
    insert zhriflog.
